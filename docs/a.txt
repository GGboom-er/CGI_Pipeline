    def log(self, msg):

    # --- Utils ---
    def get_sel(self, t):
        s = cmds.ls(os=True, fl=True, l=True)
        if not s: return []
        if t=="v": return [x for x in s if ".vtx[" in x]
        if t=="j": return cmds.ls(sl=True, type="joint", l=True)
        return []
    def store_src_vtx(self): 
        self.source_vertices = self.get_sel("v"); self.lbl_src_count.setText(str(len(self.source_vertices)))
    def store_tgt_vtx(self): 
        self.target_vertices = self.get_sel("v"); self.lbl_tgt_count.setText(str(len(self.target_vertices)))
    def update_list(self, w, is_s):
        j = self.get_sel("j"); 
        if not j: return
        w.clear(); [w.addItem(x.split("|")[-1]) for x in j]
        if is_s: self.source_joints = j
        else: self.target_joints = j
    def get_skin_cluster(self, node):
        h = cmds.listHistory(node, pdo=True) or []
        s = cmds.ls(h, type="skinCluster"); return s[0] if s else None

    # --- Data ---
    def get_world_pts(self, vtx_list):
        return np.array([cmds.xform(v, q=True, ws=True, t=True)[:3] for v in vtx_list])
    def get_matrix(self, node):
        return np.array(cmds.getAttr(f"{node}.worldMatrix[0]")).reshape(4,4)
    def to_local(self, pts_world, joint, mx=False, my=False, mz=False):
        mat = self.get_matrix(joint); inv = np.linalg.inv(mat)
        pts_4 = np.hstack([pts_world, np.ones((len(pts_world), 1))])
        local = (inv @ pts_4.T).T[:, :3]
        if mx: local[:,0]*=-1
        if my: local[:,1]*=-1
        if mz: local[:,2]*=-1
        return local
    def get_full_source_weights(self):
        mesh = self.source_vertices[0].split(".")[0]; sk=self.get_skin_cluster(mesh); 
        if not sk: return None
        sel = om.MSelectionList(); sel.add(sk); fn = oma.MFnSkinCluster(sel.getDependNode(0))
        inds = [int(v.split("[")[-1].split("]")[0]) for v in self.source_vertices]
        selm = om.MSelectionList(); selm.add(mesh); dp = selm.getDagPath(0)
        fnc = om.MFnSingleIndexedComponent(); comp = fnc.create(om.MFn.kMeshVertComponent); fnc.addElements(inds)
        w, infs = fn.getWeights(dp, comp); w=np.array(w)
        paths = [d.fullPathName() for d in fn.influenceObjects()]
        n_v = len(self.source_vertices); w_raw = w.reshape(n_v, infs)
        w_ord = np.zeros((n_v, len(self.source_joints)))
        for i, j in enumerate(self.source_joints):
            try:
                if j in paths: idx = paths.index(j)
                else: 
                    short = j.split("|")[-1]
                    for k, p in enumerate(paths):
                        if p.split("|")[-1] == short: idx = k; break
                w_ord[:, i] = w_raw[:, idx]
            except: pass
        return w_ord

    # --- ALGORITHMS ---
    def weighted_average(self, points, weights):
        if weights is None: return np.mean(points, axis=0)
        w_sum = np.sum(weights)
        if w_sum < 1e-6: return np.mean(points, axis=0)
        return np.average(points, axis=0, weights=weights)

    def best_fit_transform_weighted(self, A, B, weights=None):
        cent_A = self.weighted_average(A, weights)
        cent_B = self.weighted_average(B, weights)
        AA = A - cent_A; BB = B - cent_B
        if weights is not None:
            w_diag = np.diag(weights)
            H = np.dot(AA.T, np.dot(w_diag, BB))
        else: H = np.dot(AA.T, BB)
        U, S, Vt = np.linalg.svd(H)
        R = np.dot(Vt.T, U.T)
        if np.linalg.det(R) < 0: Vt[2,:]*=-1; R=np.dot(Vt.T, U.T)
        t = cent_B.T - np.dot(R, cent_A.T)
        return R, t

    def solve_icp_weighted(self, src, src_weights, tgt, max_iter=15, bias=True):
        src_c = np.copy(src); tgt_c = np.copy(tgt)
        w_vals = src_weights if bias else None
        if w_vals is not None:
            w_max = np.max(w_vals)
            if w_max > 0: w_vals /= w_max
        prev_err = 0
        for i in range(max_iter):
            dists = np.linalg.norm(src_c[:,None,:] - tgt_c[None,:,:], axis=2)
            indices = np.argmin(dists, axis=0)
            closest_src = src_c[indices]
            closest_weights = w_vals[indices] if w_vals is not None else None
            R, t = self.best_fit_transform_weighted(tgt_c, closest_src, closest_weights)
            tgt_c = np.dot(tgt_c, R.T) + t.T
            mean_err = np.mean(np.min(dists, axis=0))
            if abs(prev_err - mean_err) < 0.0001: break
            prev_err = mean_err
        return tgt_c

    def run_process(self):
        if not HAS_NUMPY: return cmds.error("Numpy missing")
        if not (self.source_vertices and self.target_vertices): return cmds.warning("Select Data")
        
        # Mapping
        mapped_tgt = []
        if self.rb_name.isChecked():
            for sj in self.source_joints:
                ss=sj.split("|")[-1]; best=None; br=-1.0
                for tj in self.target_joints:
                    r=difflib.SequenceMatcher(None, ss, tj.split("|")[-1]).ratio()
                    if r>br: br=r; best=tj
                mapped_tgt.append(best)
        else: mapped_tgt = self.target_joints # Ensure this is assigned!

        # Params
        mx=self.cb_mir_x.isChecked(); my=self.cb_mir_y.isChecked(); mz=self.cb_mir_z.isChecked()
        use_weight_bias = self.cb_weighted.isChecked()
        nudge_factor = self.sb_nudge.value()
        s_min = self.sb_smin.value(); s_max = self.sb_smax.value()
        max_inf = self.sb_max.value()

        # Data
        self.log("Reading..."); s_pos = self.get_world_pts(self.source_vertices); w_src = self.get_full_source_weights()
        t_pos = self.get_world_pts(self.target_vertices)
        
        # FIX: Variable name W_accum (Capital W)
        W_accum = np.zeros((len(t_pos), len(mapped_tgt)))

        self.log("Running Weighted ICP..."); cmds.waitCursor(state=True)
        try:
            for j_idx in range(len(self.source_joints)):
                sj = self.source_joints[j_idx]; tj = mapped_tgt[j_idx]
                col = w_src[:, j_idx]; mask = col > 0.001
                if not np.any(mask): continue
                
                s_pts = s_pos[mask]; s_vals = col[mask]
                s_loc = self.to_local(s_pts, sj)
                t_loc = self.to_local(t_pos, tj, mx, my, mz)

                # Pre-Align
                s_c = self.weighted_average(s_loc, s_vals if use_weight_bias else None)
                td = np.linalg.norm(t_loc, axis=1); t_mask = td < 20.0
                if np.any(t_mask): t_c = np.mean(t_loc[t_mask], axis=0)
                else: t_c = np.mean(t_loc, axis=0)
                s_centered = s_loc - s_c; t_centered = t_loc - t_c
                
                # Scale
                s_rms = np.sqrt(np.mean(np.sum(s_centered**2, axis=1)))
                t_rms = np.sqrt(np.mean(np.sum(t_centered**2, axis=1)))
                scale = 1.0
                if t_rms > 0:
                    raw = s_rms / t_rms
                    scale = max(s_min, min(s_max, raw))
                    if self.cb_debug.isChecked(): self.log(f"Bone {j_idx}: Scale {raw:.3f} -> {scale:.3f}")
                
                t_aligned = t_centered * scale + s_c 
                
                # ICP
                if len(s_loc) > 500:
                    idxr = np.random.choice(len(s_loc), 500, replace=False)
                    s_icp = s_loc[idxr]; w_icp = s_vals[idxr]
                else: s_icp = s_loc; w_icp = s_vals
                
                t_final = self.solve_icp_weighted(s_icp, w_icp, t_aligned, max_iter=15, bias=use_weight_bias)
                
                # Nudge
                if nudge_factor > 0.01:
                    dists = np.linalg.norm(s_icp[:,None,:] - t_final[None,:,:], axis=2)
                    near_idx = np.argmin(dists, axis=0)
                    near_pts = s_icp[near_idx]
                    t_final = t_final * (1.0 - nudge_factor) + near_pts * nudge_factor
                
                # TPS RBF
                if len(s_loc) > 600:
                    idxr = np.random.choice(len(s_loc), 600, replace=False)
                    s_sol = s_loc[idxr]; v_sol = s_vals[idxr]
                else: s_sol = s_loc; v_sol = s_vals
                
                d_ss = np.linalg.norm(s_sol[:,None,:]-s_sol[None,:,:], axis=2)
                K_ss = np.zeros_like(d_ss); m=d_ss>1e-9; K_ss[m]=(d_ss[m]**2)*np.log(d_ss[m])
                K_ss[np.diag_indices_from(K_ss)]+=0.5
                try: w_cof = np.linalg.solve(K_ss, v_sol)
                except: K_ss[np.diag_indices_from(K_ss)]+=1.0; w_cof=np.linalg.solve(K_ss, v_sol)
                
                d_ts = np.linalg.norm(t_final[:,None,:]-s_sol[None,:,:], axis=2)
                K_ts = np.zeros_like(d_ts); m=d_ts>1e-9; K_ts[m]=(d_ts[m]**2)*np.log(d_ts[m])
                pred = K_ts @ w_cof
                
                # Cleanup
                s_cnt = np.mean(s_sol, axis=0)
                dc = np.linalg.norm(t_final - s_cnt, axis=1)
                rmax = np.max(np.linalg.norm(s_sol - s_cnt, axis=1))
                valid = dc < (rmax * 2.2)
                W_accum[:, j_idx] = np.clip(pred, 0, 1) * valid

            # Apply
            self.log("Writing...")
            for i in range(len(W_accum)):
                row = W_accum[i]
                if np.count_nonzero(row)>max_inf:
                    idx=np.argsort(row)[-max_inf:]; nr=np.zeros_like(row); nr[idx]=row[idx]; row=nr
                s=np.sum(row); 
                if s>0: row/=s
                W_accum[i] = row
            
            tm={}
            for i,v in enumerate(self.target_vertices):
                m=v.split(".")[0]
                if m not in tm: tm[m]=[]
                tm[m].append(i)
                
            for m, idxs in tm.items():
                sk=self.get_skin_cluster(m)
                if not sk: continue
                cmds.setAttr(f"{sk}.maintainMaxInfluences", 0); cmds.setAttr(f"{sk}.normalizeWeights", 0)
                ex=set(cmds.skinCluster(sk,q=True,inf=True) or [])
                
                # FIX: use mapped_tgt here!
                for j in mapped_tgt:
                    if j and j not in ex:
                        try: cmds.skinCluster(sk,e=True,ai=j,lw=True,wt=0); cmds.setAttr(f"{j}.locked", 0)
                        except: pass
                
                for i in idxs:
                    r=W_accum[i]
                    # FIX: use mapped_tgt here!
                    v=[(mapped_tgt[k], w) for k,w in enumerate(r) if w>0.001]
                    if v: cmds.skinPercent(sk, self.target_vertices[i], transformValue=v, normalize=False, zeroRemainingInfluences=True)
                cmds.setAttr(f"{sk}.normalizeWeights", 1); cmds.setAttr(f"{sk}.maxInfluences", max_inf)
                cmds.skinPercent(sk, [self.target_vertices[i] for i in idxs], normalize=True)
            cmds.inViewMessage(amg="V13.1 Fixed Complete", pos='midCenter', fade=True)
        except Exception as e: cmds.error(f"{e}"); import traceback; traceback.print_exc()
        finally: cmds.waitCursor(state=False)
