import maya.cmds as cmds
import maya.OpenMaya as om
import maya.OpenMayaAnim as oma

class BatchWeightCopyTool:
    def __init__(self):
        self.window_name = "batchWeightCopyWindow"
        self.create_ui()

    def create_ui(self):
        if cmds.window(self.window_name, exists=True):
            cmds.deleteUI(self.window_name)

        cmds.window(self.window_name, title="批量权重发散性拷贝工具", width=300)
        cmds.columnLayout(adjustableColumn=True)

        cmds.text(label="选择源点:")
        self.source_field = cmds.textField()
        cmds.button(label="获取选择的源点", command=self.get_source_selection)

        cmds.text(label="选择目标点:")
        self.target_field = cmds.textField()
        cmds.button(label="获取选择的目标点", command=self.get_target_selection)

        cmds.text(label="影响半径:")
        self.radius_field = cmds.floatFieldGrp(numberOfFields=1, value1=1.0)

        cmds.checkBox(label="只拷贝最大权重", value=True, changeCommand=self.toggle_max_weight)
        self.max_weight_only = True

        cmds.checkBox(label="强制更新权重", value=False, changeCommand=self.toggle_force_update)
        self.force_update = False

        cmds.button(label="执行权重拷贝", command=self.execute_weight_copy)

        cmds.showWindow(self.window_name)

    def get_source_selection(self, *args):
        selection = cmds.ls(selection=True, flatten=True)
        cmds.textField(self.source_field, edit=True, text=",".join(selection))

    def get_target_selection(self, *args):
        selection = cmds.ls(selection=True, flatten=True)
        cmds.textField(self.target_field, edit=True, text=",".join(selection))

    def toggle_max_weight(self, value):
        self.max_weight_only = value

    def toggle_force_update(self, value):
        self.force_update = value

    def get_skin_cluster(self, vertex):
        mesh = vertex.split('.')[0]
        skin_clusters = cmds.ls(cmds.listHistory(mesh), type='skinCluster')
        if skin_clusters:
            return skin_clusters[0]
        return None

    def execute_weight_copy(self, *args):
        source_points = cmds.textField(self.source_field, query=True, text=True).split(",")
        target_points = cmds.textField(self.target_field, query=True, text=True).split(",")
        radius = cmds.floatFieldGrp(self.radius_field, query=True, value1=True)

        if not source_points or not target_points:
            cmds.warning("请确保已选择源点和目标点")
            return

        skin_cluster = self.get_skin_cluster(source_points[0])
        if not skin_cluster:
            cmds.warning("未找到蒙皮变形器，请确保选择的点属于蒙皮物体")
            return

        # 获取源点的权重
        source_weights = self.get_weights(source_points, skin_cluster)

        # 计算并应用目标点的权重
        success = self.apply_weights(source_points, target_points, source_weights, radius, skin_cluster)

        if success:
            cmds.inViewMessage(amg="权重拷贝完成", pos='midCenter', fade=True)
        else:
            cmds.warning("权重拷贝过程中出现错误，请检查脚本编辑器中的详细信息")

    def get_weights(self, points, skin_cluster):
        weights = {}
        skin_cluster_fn = oma.MFnSkinCluster(self.get_mobject(skin_cluster))
        
        # 获取所有影响器
        influence_objects = om.MDagPathArray()
        skin_cluster_fn.influenceObjects(influence_objects)
        
        for point in points:
            dag_path, component = self.get_dag_path_and_component(point)
            influence_indices = om.MIntArray()
            influence_weights = om.MDoubleArray()
            
            # 获取所有影响器的索引
            for i in range(influence_objects.length()):
                influence_indices.append(i)
            
            try:
                skin_cluster_fn.getWeights(dag_path, component, influence_indices, influence_weights)
                weights[point] = dict(zip(range(influence_indices.length()), influence_weights))
                print(f"源点 {point} 的权重: {weights[point]}")
            except Exception as e:
                print(f"获取 {point} 的权重时出错: {str(e)}")
                weights[point] = {}

        return weights

    def apply_weights(self, source_points, target_points, source_weights, radius, skin_cluster):
        skin_cluster_fn = oma.MFnSkinCluster(self.get_mobject(skin_cluster))

        # 获取所有影响器
        influence_objects = om.MDagPathArray()
        skin_cluster_fn.influenceObjects(influence_objects)
        influence_indices = om.MIntArray()
        for i in range(influence_objects.length()):
            influence_indices.append(i)

        for target in target_points:
            closest_source = self.find_closest_source(target, source_points, radius)
            if closest_source:
                dag_path, component = self.get_dag_path_and_component(target)
                try:
                    self.set_weight(skin_cluster_fn, dag_path, component, source_weights[closest_source], target, influence_indices)
                    print(f"成功应用权重到 {target}")
                except Exception as e:
                    print(f"应用权重到 {target} 时出错: {str(e)}")
                    return False
            else:
                print(f"没有找到 {target} 的最近源点")

        return True

    def find_closest_source(self, target, source_points, radius):
        target_point = om.MPoint(*cmds.xform(target, query=True, translation=True, worldSpace=True))
        closest_distance = float('inf')
        closest_source = None

        for source in source_points:
            source_point = om.MPoint(*cmds.xform(source, query=True, translation=True, worldSpace=True))
            distance = target_point.distanceTo(source_point)
            if distance <= radius and distance < closest_distance:
                closest_distance = distance
                closest_source = source

        return closest_source

    def set_weight(self, skin_cluster_fn, dag_path, component, weights, target, influence_indices):
        influence_weights = om.MDoubleArray()

        if self.max_weight_only:
            max_influence = max(weights, key=weights.get)
            for i in range(influence_indices.length()):
                influence_weights.append(1.0 if i == max_influence else 0.0)
        else:
            for i in range(influence_indices.length()):
                influence_weights.append(weights.get(i, 0.0))

        old_weights = om.MDoubleArray()
        skin_cluster_fn.getWeights(dag_path, component, influence_indices, old_weights)
        
        print(f"目标点 {target} 的旧权重: {list(old_weights)}")
        print(f"准备设置的新权重: {list(influence_weights)}")

        skin_cluster_fn.setWeights(dag_path, component, influence_indices, influence_weights, False)

        new_weights = om.MDoubleArray()
        skin_cluster_fn.getWeights(dag_path, component, influence_indices, new_weights)

        print(f"目标点 {target} 的新权重: {list(new_weights)}")

        if not self.compare_weights(old_weights, new_weights) or self.force_update:
            print(f"权重已成功更新: 旧权重 = {list(old_weights)}, 新权重 = {list(new_weights)}")
        else:
            raise Exception("权重未发生变化")

    def compare_weights(self, weights1, weights2):
        if weights1.length() != weights2.length():
            return False
        return all(abs(weights1[i] - weights2[i]) < 1e-5 for i in range(weights1.length()))

    def get_mobject(self, node_name):
        selection_list = om.MSelectionList()
        selection_list.add(node_name)
        mobject = om.MObject()
        selection_list.getDependNode(0, mobject)
        return mobject

    def get_dag_path_and_component(self, vertex):
        selection_list = om.MSelectionList()
        selection_list.add(vertex)
        dag_path = om.MDagPath()
        component = om.MObject()
        selection_list.getDagPath(0, dag_path, component)
        return dag_path, component

# 创建并显示工具窗口
tool = BatchWeightCopyTool()
