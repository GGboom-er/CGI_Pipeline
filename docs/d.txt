class CopyTheWeightJBY:
    def __init__(self):
        self.source_skin_node = None
        self.bones = []
        self.target_objects = []

    @staticmethod
    def get_all_history_nodes_by_type(input_nodes, node_type):
        all_nodes = set()
        for node in input_nodes:
            history_nodes = cmds.listHistory(node)
            if history_nodes:
                nodes_of_type = cmds.ls(history_nodes, type=node_type)
                all_nodes.update(nodes_of_type)
        return list(all_nodes)

    def create_skin_cluster_to_objects(self):
        for obj in self.target_objects:
            existing_skin_clusters = self.get_all_history_nodes_by_type([obj], 'skinCluster')
            if existing_skin_clusters:
                new_skin_cluster = existing_skin_clusters[0]
            else:
                new_skin_cluster = cmds.skinCluster(self.bones, obj, tsb=True)[0]
            cmds.copySkinWeights(ss=self.source_skin_node, ds=new_skin_cluster, nm=True, sa='closestPoint', ia=['oneToOne', 'closestJoint', 'closestJoint'])

    @staticmethod
    def get_all_children(inputs):
        outputs = []
        nodes_to_process = inputs[:]
        while nodes_to_process:
            item = nodes_to_process.pop(0)
            if cmds.listRelatives(item, shapes=True):
                outputs.append(item)
            children = cmds.listRelatives(item, f=True, type='transform', c=True)
            if children:
                nodes_to_process.extend(children)
        return outputs

    def start_copy_weight(self):
        selection = cmds.ls(sl=True)
        if not selection:
            raise ValueError("Please select the source model and target models first!")

        self.bones = self.get_all_history_nodes_by_type(selection[:1], 'joint')
        if not self.bones:
            raise ValueError("The first selected source object has no skin!")

        skin_clusters = self.get_all_history_nodes_by_type(selection[:1], 'skinCluster')
        if not skin_clusters:
            raise ValueError("The first selected source object has no skin!")

        self.source_skin_node = skin_clusters[0]
        self.target_objects = self.get_all_children(selection[1:])
        self.create_skin_cluster_to_objects()

class AdvancedCopyWeight:
    def __init__(self):
        self.source_meshes = {}  # {short_name: full_name}
        self.target_meshes = {}  # {short_name: full_name}
        self.matched_pairs = []  # [(source_full_name, target_full_name)]
        self.unmatched_sources = []
        self.unmatched_targets = []
        self.copy_results = {}  # {short_name: (status, message)}
        self.replace_from = ""  # Source string, e.g. "low"
        self.replace_to = ""    # Target string, e.g. "high"

    @staticmethod
    def get_short_name(full_name):
        """Get object's short name (without path and common suffixes)"""
        short_name = full_name.split('|')[-1].split(':')[-1]
        # Remove common suffixes like _target, _copy, _duplicate
        suffixes = ['_target', '_copy', '_duplicate', '_new', '_copy1', '_copy2']
        for suffix in suffixes:
            if short_name.endswith(suffix):
                short_name = short_name[:-len(suffix)]
                break
        return short_name

    def set_replace_strings(self, replace_from, replace_to):
        """Set character replacement rules"""
        self.replace_from = replace_from.strip()
        self.replace_to = replace_to.strip()

    def normalize_name_for_matching(self, name):
        """Normalize name for matching, apply character replacement rules"""
        if self.replace_from and self.replace_to:
            # If replacement rules are set, replace source string with target string
            if self.replace_from in name:
                return name.replace(self.replace_from, self.replace_to)
        return name

    def get_meshes_from_group(self, group_name):
        """Get all MESH objects from group, including nested groups"""
        meshes = {}
        if not cmds.objExists(group_name):
            return meshes
        
        # Recursively get all child groups and objects
        def get_all_transforms_recursive(node):
            transforms = []
            
            # Get direct children
            children = cmds.listRelatives(node, children=True, type='transform', fullPath=True) or []
            
            for child in children:
                # Check if has mesh shape
                shapes = cmds.listRelatives(child, shapes=True, type='mesh', fullPath=True)
                if shapes:
                    transforms.append(child)
                
                # Recursively process child groups
                grandchildren = cmds.listRelatives(child, children=True, type='transform')
                if grandchildren:
                    transforms.extend(get_all_transforms_recursive(child))
            
            return transforms
        
        # Get all transform nodes
        all_transforms = get_all_transforms_recursive(group_name)
        
        # Build mapping from short name to full name
        for transform in all_transforms:
            short_name = self.get_short_name(transform)
            # If duplicate names, use more complete path
            if short_name in meshes:
                # Generate more unique name
                parent_path = transform.split('|')[-2] if '|' in transform else ""
                unique_name = f"{parent_path}_{short_name}" if parent_path else short_name
                meshes[unique_name] = transform
            else:
                meshes[short_name] = transform
                
        return meshes

    def analyze_groups(self, source_group, target_group):
        """Analyze source and target groups to find matching and non-matching meshes"""
        self.source_meshes = self.get_meshes_from_group(source_group)
        self.target_meshes = self.get_meshes_from_group(target_group)
        
        self.matched_pairs = []
        self.unmatched_sources = []
        self.unmatched_targets = []
        self.copy_results = {}
        
        # Find matching pairs
        for short_name, source_full in self.source_meshes.items():
            # Normalize source name for matching
            normalized_source_name = self.normalize_name_for_matching(short_name)
            
            # Find matching target
            matched_target = None
            matched_target_name = None
            
            for target_short_name, target_full in self.target_meshes.items():
                # Normalize target name for matching
                normalized_target_name = self.normalize_name_for_matching(target_short_name)
                
                # Check if matches
                if normalized_source_name == normalized_target_name:
                    matched_target = target_full
                    matched_target_name = target_short_name
                    break
            
            if matched_target:
                self.matched_pairs.append((source_full, matched_target))
            else:
                self.unmatched_sources.append(short_name)
        
        # Find unmatched in target group
        matched_target_names = [self.get_short_name(target_full) for _, target_full in self.matched_pairs]
        for target_short_name in self.target_meshes:
            if target_short_name not in matched_target_names:
                self.unmatched_targets.append(target_short_name)

    def copy_weights_between_pairs(self):
        """Copy weights between matched pairs"""
        self.copy_results = {}
        
        for source_full, target_full in self.matched_pairs:
            short_name = self.get_short_name(source_full)
            try:
                # Get source skinCluster
                source_skin_clusters = CopyTheWeightJBY.get_all_history_nodes_by_type([source_full], 'skinCluster')
                if not source_skin_clusters:
                    self.copy_results[short_name] = ("failed", f"Source object has no skin")
                    continue
                
                source_skin = source_skin_clusters[0]
                
                # Get bones
                bones = CopyTheWeightJBY.get_all_history_nodes_by_type([source_full], 'joint')
                if not bones:
                    self.copy_results[short_name] = ("failed", f"Source object has no joints")
                    continue
                
                # Check if target already has skinCluster
                target_skin_clusters = CopyTheWeightJBY.get_all_history_nodes_by_type([target_full], 'skinCluster')
                if target_skin_clusters:
                    target_skin = target_skin_clusters[0]
                else:
                    # Create new skinCluster
                    target_skin = cmds.skinCluster(bones, target_full, tsb=True)[0]
                
                # Copy weights
                cmds.copySkinWeights(ss=source_skin, ds=target_skin, nm=True, sa='closestPoint', 
                                   ia=['oneToOne', 'closestJoint', 'closestJoint'])
                
                self.copy_results[short_name] = ("success", "Success")
                
            except Exception as e:
                self.copy_results[short_name] = ("failed", f"Error: {str(e)}")
        
        return self.copy_results

    def copy_weights(self, *args):
        """Execute weight copy"""
        try:
            if not self.copy_weight_tool.matched_pairs:
                cmds.confirmDialog(title='Error', message='No matched mesh pairs! Please analyze groups first.', button=['OK'])
                return
            
            # Clear results list
            cmds.textScrollList(self.result_list, edit=True, removeAll=True)
            cmds.textScrollList(self.result_list, edit=True, append="Copying weights...")
            
            # Execute weight copy
            results = self.copy_weight_tool.copy_weights_between_pairs()
            
            # Update results list
            cmds.textScrollList(self.result_list, edit=True, removeAll=True)
            
            success_count = 0
            failed_count = 0
            
            for short_name, (status, message) in results.items():
                if status == "success":
                    cmds.textScrollList(self.result_list, edit=True, append=f"✓ {short_name}: Success")
                    success_count += 1
                else:
                    cmds.textScrollList(self.result_list, edit=True, append=f"✗ {short_name}: {message}")
                    failed_count += 1
            
            # Show summary
            result_message = f"Weight copy complete!\n\nSuccess: {success_count}\nFailed: {failed_count}"
            
            if failed_count > 0:
                result_message += f"\n\nFailed items marked with ✗ in results list"
            
            cmds.confirmDialog(title='Copy Results', message=result_message, button=['OK'])
            
        except Exception as e:
            cmds.confirmDialog(title='Error', message=f'Error during weight copy: {str(e)}', button=['OK'])
