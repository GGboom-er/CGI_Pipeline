CACHED_DATA = {}
# Global variables to store source vertex selection state
SOURCE_SELECTION_STATE = {
    'mesh': None,
    'indices': []
}

def get_maya_main_window():
    """Returns the QWidget object for the Maya main window."""
    try:
        main_window_ptr = omui.MQtUtil.mainWindow()
        if main_window_ptr is not None:
            return shiboken2.wrapInstance(int(main_window_ptr), QWidget)
    except Exception as e:
        cmds.warning(f"Could not get Maya main window handle: {e}")
        return None

def get_selection_info():
    """
    Get the selected object (mesh transform) and selected vertex indices.
    Always returns the Long Name (Full Path) to avoid ambiguity.
    """
    selected = cmds.ls(selection=True, flatten=True, long=True)
    if not selected:
        return None, None

    # Determine if vertices are selected
    vertex_selections = [s for s in selected if '.vtx' in s]
    
    if vertex_selections:
        # Vertex selection mode
        # Extracts mesh name from "group|mesh.vtx[1]" -> "group|mesh"
        mesh_name = vertex_selections[0].split(".vtx")[0]
        selected_indices = [int(vtx.split("[")[-1].rstrip("]")) for vtx in vertex_selections]
        return mesh_name, selected_indices
    else:
        # Object selection mode (take the first selected object)
        mesh_name = selected[0]
        # Check if it's a valid mesh transform
        shapes = cmds.listRelatives(mesh_name, shapes=True, fullPath=True, type='mesh')
        if not shapes:
            # Try to see if the selection itself is a shape
            if cmds.nodeType(mesh_name) == 'mesh':
                mesh_name = cmds.listRelatives(mesh_name, parent=True, fullPath=True)[0]
            else:
                cmds.warning(f"'{mesh_name}' is not a valid mesh transform.")
                return None, None
            
        # For object mode, indices will be determined inside transfer_weights_main
        return mesh_name, None

def record_source_vertices():
    """Records selected vertices from source model, updating the global state."""
    mesh_name, selected_indices = get_selection_info()
    
    if not mesh_name or not selected_indices:
        cmds.warning("Please select vertices on the source model first!")
        return False

    # Check if a skin cluster exists, otherwise transfer won't work
    skin_cluster = cmds.ls(cmds.listHistory(mesh_name), type="skinCluster")
    if not skin_cluster:
        cmds.warning(f"Source mesh '{mesh_name}' has no skin cluster. Cannot record.")
        return False
        
    global SOURCE_SELECTION_STATE
    SOURCE_SELECTION_STATE['mesh'] = mesh_name
    SOURCE_SELECTION_STATE['indices'] = selected_indices

    # Show short name in message for readability
    short_name = mesh_name.split('|')[-1]
    cmds.inViewMessage(
        amg=f"Recorded source vertices on {short_name} ({len(selected_indices)} vertices)", 
        pos="midCenter", 
        fade=True
    )
    return True

def get_mesh_data(mesh_name):
    """Fetches and caches core mesh data (vertices, normals, curvature, adjacency, kdtree)."""
    if mesh_name in CACHED_DATA and all(k in CACHED_DATA[mesh_name] for k in ['vertices', 'normals', 'curvature', 'features', 'adjacency']):
        return CACHED_DATA[mesh_name]

    try:
        selection_list = om.MSelectionList()
        selection_list.add(mesh_name)
        dag_path = selection_list.getDagPath(0)
        mesh_fn = om.MFnMesh(dag_path)

        # 1. Vertices and Normals
        vertices = mesh_fn.getPoints(om.MSpace.kWorld)
        normals = mesh_fn.getVertexNormals(angleWeighted=True)
        vertices_np = np.array([(v.x, v.y, v.z) for v in vertices])
        normals_np = np.array([(n.x, n.y, n.z) for n in normals])
        
        if len(vertices_np) == 0:
            cmds.warning(f"Mesh '{mesh_name}' has no vertices.")
            return None

        # 2. Adjacency 
        adjacency = {}
        for i in range(mesh_fn.numVertices):
            adjacency[i] = set()
        for face_id in range(mesh_fn.numPolygons):
            vtx_indices = mesh_fn.getPolygonVertices(face_id)
            for i in range(len(vtx_indices)):
                for j in range(i + 1, len(vtx_indices)):
                    adjacency[vtx_indices[i]].add(vtx_indices[j])
                    adjacency[vtx_indices[j]].add(vtx_indices[i])
        adjacency = {k: list(v) for k, v in adjacency.items()}
        
        # 3. Curvature (Approximation using Laplacian)
        curvature = np.zeros(mesh_fn.numVertices)
        for i in range(mesh_fn.numVertices):
            if i in adjacency and adjacency[i]:
                neighbor_positions = np.array([(vertices[j].x, vertices[j].y, vertices[j].z) for j in adjacency[i] if j < mesh_fn.numVertices])
                if len(neighbor_positions) > 0:
                    center = vertices_np[i]
                    mean_position = np.mean(neighbor_positions, axis=0)
                    curvature[i] = np.linalg.norm(center - mean_position)
        max_curv = np.max(curvature)
        if max_curv > 0:
            curvature /= max_curv
        
        # 4. Features (Curvature-weighted normals)
        features = normals_np * (1.0 + curvature.reshape(-1, 1) * 2.0)

        # Cache the data
        CACHED_DATA[mesh_name] = {
            'vertices': vertices_np,
            'normals': normals_np,
            'curvature': curvature,
            'features': features,
            'adjacency': adjacency,
            'kdtree': cKDTree(vertices_np)
        }
        return CACHED_DATA[mesh_name]

    except Exception as e:
        cmds.warning(f"Error getting mesh data for {mesh_name}: {e}")
        return None

def get_closest_vertices_simplified(source_data, target_pos, target_normal, target_curv, k=8, fidelity=1.0, source_indices=None):
    """Find k closest source vertices using spatial and feature-based matching, influenced by Fidelity."""
    
    # Filter source data if specific indices are provided (Vertex-to-Vertex mode)
    if source_indices is not None and len(source_indices) > 0:
        # Create temporary, filtered data for KD-Tree search
        filtered_vertices = source_data['vertices'][source_indices]
        # Note: Rebuilding KDTree is expensive, but necessary for sparse selections
        filtered_kdtree = cKDTree(filtered_vertices)
        
        # Get candidates relative to the filtered set
        candidate_count = min(k * 3, len(filtered_vertices))
        if candidate_count == 0: return np.array([])
        
        distances, relative_candidate_indices = filtered_kdtree.query(target_pos, k=candidate_count)
        
        # Map relative indices back to original indices
        candidate_indices = np.array([source_indices[i] for i in relative_candidate_indices])
    else:
        # Full mesh search mode
        filtered_vertices = source_data['vertices']
        filtered_kdtree = source_data['kdtree']
        candidate_count = min(k * 3, len(filtered_vertices))
        if candidate_count == 0: return np.array([])
        
        distances, candidate_indices = filtered_kdtree.query(target_pos, k=candidate_count)

    if np.isscalar(candidate_indices):
        candidate_indices = np.array([candidate_indices])
        distances = np.array([distances])
    
    if len(candidate_indices) == 0:
        return np.array([])

    normals = source_data['normals']
    curvature = source_data['curvature']

    # Fidelity-driven Weights
    base_normal_w = 0.2 + (fidelity * 0.5) 
    base_curvature_w = 0.1 + (fidelity * 0.4) 
    
    scores = np.zeros(len(candidate_indices))
    
    for i, idx in enumerate(candidate_indices):
        position_score = distances[i] * 1.0 
        
        normal_similarity = np.dot(normals[idx], target_normal)
        normal_score = (1.0 - normal_similarity) * base_normal_w
        
        curvature_score = abs(curvature[idx] - target_curv) * base_curvature_w
        
        scores[i] = position_score + normal_score + curvature_score
    
    best_k_indices = np.argsort(scores)[:k]
    return candidate_indices[best_k_indices]

# Smooth Weights 
# ====================================================================
def anisotropic_smooth_weights_simplified(weights_dict, adjacency_list, target_curvature, smoothness=0.3, target_indices=None):
    """Smooth weights using an iterative Laplacian approach."""
    if not weights_dict or smoothness <= 0.001:
        return weights_dict
    
    iterations = 2 if smoothness < 0.5 else 4
    alpha = smoothness * 0.8 

    for _ in range(iterations):
        new_weights_dict = {}
        
        for i, current_weights in weights_dict.items():
            
            if i not in adjacency_list or not adjacency_list[i]:
                new_weights_dict[i] = current_weights
                continue

            neighbor_weights_accum = []
            
            for neighbor in adjacency_list[i]:
                # Check constraints: 
                if neighbor in weights_dict:
                    if target_indices is None or neighbor in target_indices:
                        neighbor_weights_accum.append(weights_dict[neighbor])
            
            if not neighbor_weights_accum:
                new_weights_dict[i] = current_weights
                continue
            
            avg_neighbor_weights = np.mean(neighbor_weights_accum, axis=0)
            
            local_alpha = alpha
            if target_curvature is not None and i < len(target_curvature):
                local_alpha *= (1.0 - target_curvature[i] * 0.5)

            smoothed_result = (1.0 - local_alpha) * current_weights + local_alpha * avg_neighbor_weights
            
            if np.sum(smoothed_result) > 0:
                smoothed_result /= np.sum(smoothed_result)
            
            new_weights_dict[i] = smoothed_result
        
        weights_dict = new_weights_dict
    
    return weights_dict

def interpolate_weights(source_weights, distances, fidelity):
    """Interpolate weights based on distance and fidelity."""
    if len(source_weights) == 0 or len(distances) == 0:
        return np.array([])

    source_weights = np.array(source_weights)
    
    min_rows = min(len(distances), source_weights.shape[0])
    if min_rows == 0:
        return np.zeros(source_weights.shape[1] if len(source_weights.shape) > 1 else 0)
    
    distances = distances[:min_rows]
    source_weights = source_weights[:min_rows]
    
    fit_factor = 1.0 + (fidelity * 3.0) 

    weights = 1.0 / (distances + 1e-10)
    weights = np.power(weights, fit_factor)
    
    weights_sum = np.sum(weights)
    if weights_sum > 0:
        weights = weights / weights_sum
    else:
        return np.zeros_like(source_weights[0])
    
    weights = weights.reshape(-1, 1)
    interpolated = np.sum(weights * source_weights, axis=0)
    
    if np.sum(interpolated) > 0:
        interpolated /= np.sum(interpolated)
    else:
        interpolated = np.zeros_like(interpolated)
        
    return interpolated

def transfer_weights_main(source_mesh, target_mesh, fidelity, smoothness, max_influences, mode='mesh'):
    """The main weight transfer logic."""
    
    global SOURCE_SELECTION_STATE
    
    source_indices = None
    target_indices = None
    
    # --- FIX: Robust Mode Handling ---
    if mode == 'vertex':
        # 1. Source Validation
        source_indices = SOURCE_SELECTION_STATE['indices']
        if not source_indices:
             cmds.error("Vertex transfer mode requires source vertices to be recorded.")
             return
        
        # 2. Target Validation (The Fix for 'Selected target vertices are not on mesh')
        # Instead of trusting the text field strictly, we trust the CURRENT SELECTION.
        sel_mesh, sel_indices = get_selection_info()
        
        if not sel_mesh or not sel_indices:
             cmds.error("Vertex transfer mode requires TARGET vertices to be selected in the viewport.")
             return
             
        # Compare long names to check if they are conceptually the same object
        # If the UI text differs from selection (e.g. short name vs long name), we prefer the selection.
        try:
            # Ensure we use the long name for the actual processing
            target_mesh = cmds.ls(sel_mesh, long=True)[0]
            target_indices = sel_indices
            print(f"Vertex Mode: Transferring to selected mesh '{target_mesh}' on {len(target_indices)} vertices.")
        except Exception as e:
             cmds.error(f"Error validating target selection: {e}")
             return
             
    else:
        # Mesh Mode: Just ensure we have the long name of the target
        if cmds.objExists(target_mesh):
            target_mesh = cmds.ls(target_mesh, long=True)[0]
        
    # --- End Fix ---

    # 1. Get Skin Cluster and Joint Info
    source_skin_cluster = cmds.ls(cmds.listHistory(source_mesh), type="skinCluster")
    if not source_skin_cluster: cmds.error(f"Source mesh '{source_mesh}' has no skin cluster!")
    source_skin_cluster = source_skin_cluster[0]
    source_joints = cmds.skinCluster(source_skin_cluster, query=True, influence=True)

    target_skin_cluster = cmds.ls(cmds.listHistory(target_mesh), type="skinCluster")
    if not target_skin_cluster:
        try:
            target_skin_cluster = cmds.skinCluster(source_joints, target_mesh, toSelectedBones=True, normalizeWeights=1, maximumInfluences=max_influences)[0]
        except Exception as e: cmds.error(f"Failed to create skin cluster for '{target_mesh}': {e}")
    else:
        target_skin_cluster = target_skin_cluster[0]
    target_joints = cmds.skinCluster(target_skin_cluster, query=True, influence=True)
    
    # Filter valid joints
    valid_joints = [j for j in target_joints if j in source_joints]
    # If no common joints, maybe target has none? Add source joints to target.
    if not valid_joints:
        cmds.warning("Target skin cluster shares no joints with source. Adding source joints to target.")
        for jnt in source_joints:
            try:
                cmds.skinCluster(target_skin_cluster, edit=True, addInfluence=jnt, weight=0)
            except: pass
        target_joints = cmds.skinCluster(target_skin_cluster, query=True, influence=True)

    source_joint_indices = {j: i for i, j in enumerate(source_joints)}
    target_joint_indices = {j: i for i, j in enumerate(target_joints)}
    
    # 2. Get Mesh Data
    source_data = get_mesh_data(source_mesh)
    target_data = get_mesh_data(target_mesh)
    if not source_data or not target_data: return

    source_vertices = source_data['vertices']
    target_vertices = target_data['vertices']
    target_normals = target_data['normals']
    target_curvature = target_data['curvature']
    adjacency = target_data['adjacency']
    
    # Determine which vertices to process
    if mode == 'mesh':
        total_vertices = len(target_vertices)
        indices_to_process = range(total_vertices)
    else: # mode == 'vertex'
        indices_to_process = target_indices
        total_vertices = len(indices_to_process)
        
    # 3. Process Vertices
    search_k = max(8, int(max_influences * 1.5))
    target_weights = {}
    source_weight_cache = {}
    
    cmds.progressWindow(title=f"Transferring Weights", progress=0, status="Starting...", isInterruptable=True)

    for i_count, i in enumerate(indices_to_process):
        if cmds.progressWindow(query=True, isCancelled=True):
            cmds.progressWindow(endProgress=True)
            cmds.warning("Transfer cancelled by user.")
            return

        if i_count % 50 == 0:
             progress = int(i_count / total_vertices * 100)
             cmds.progressWindow(edit=True, progress=progress, status=f"Matching vertex {i_count+1}/{total_vertices}")

        # Safety check for index
        if i >= len(target_vertices): continue
        
        pos = target_vertices[i]
        normal = target_normals[i]
        target_curv = target_curvature[i]
        
        # Matching (Fidelity) - Point to Point Logic
        closest_indices = get_closest_vertices_simplified(
            source_data, pos, normal, target_curv, 
            k=search_k, 
            fidelity=fidelity, 
            source_indices=source_indices # Passes only the recorded source points
        )
        if closest_indices.size == 0: continue

        # (Weight fetching and mapping logic)
        source_vertex_weights = []
        valid_indices = []
        
        for idx in closest_indices:
            if idx not in source_weight_cache:
                vtx = f"{source_mesh}.vtx[{idx}]"
                try:
                    # Query weights
                    raw_weights = cmds.skinPercent(source_skin_cluster, vtx, query=True, value=True)
                    transform_values = cmds.skinPercent(source_skin_cluster, vtx, query=True, transform=None)
                    source_weight_cache[idx] = (raw_weights, transform_values)
                except Exception as e:
                    print(f"Error getting weights for vertex {vtx}: {e}")
                    continue
            
            raw_weights, transform_values = source_weight_cache[idx]
            
            # Map source weights to target skeleton structure
            target_weight_vector = np.zeros(len(target_joints))
            for j, joint in enumerate(transform_values):
                if joint in target_joints:
                    target_idx = target_joint_indices[joint]
                    target_weight_vector[target_idx] = raw_weights[j]
            
            if np.sum(target_weight_vector) > 0:
                target_weight_vector /= np.sum(target_weight_vector)
                source_vertex_weights.append(target_weight_vector)
                valid_indices.append(idx)
        
        if not source_vertex_weights: continue

        distances = np.linalg.norm(source_vertices[valid_indices] - pos, axis=1)
        
        # Interpolate 
        interpolated_weights = interpolate_weights(source_vertex_weights, distances, fidelity)
        
        if len(interpolated_weights) == 0 or np.sum(interpolated_weights) == 0: continue

        # Limit influences
        sorted_indices = np.argsort(interpolated_weights)[::-1]
        limited_weights = np.zeros_like(interpolated_weights)
        
        limit_count = min(int(max_influences), len(sorted_indices))
        
        for j in range(limit_count):
            idx = sorted_indices[j]
            if interpolated_weights[idx] > 0:
                limited_weights[idx] = interpolated_weights[idx]
        
        if np.sum(limited_weights) > 0:
            limited_weights /= np.sum(limited_weights)
            target_weights[i] = limited_weights

    cmds.progressWindow(endProgress=True)

    # 4. Smooth Weights
    if smoothness > 0.0 and target_weights:
        cmds.inViewMessage(amg=f"Smoothing weights...", pos="midCenter", fade=True)
        cmds.progressWindow(title=f"Smoothing Weights", progress=0, status="Starting...", isInterruptable=True)

        target_weights = anisotropic_smooth_weights_simplified(
            target_weights, 
            adjacency, 
            target_curvature,
            smoothness=smoothness,
            target_indices=set(target_indices) if mode == 'vertex' else None
        )
        cmds.progressWindow(endProgress=True)

    # 5. Apply Weights
    cmds.inViewMessage(amg=f"Applying weights...", pos="midCenter", fade=True)
    cmds.progressWindow(title=f"Applying Weights", progress=0, status="Starting...", isInterruptable=True)

    vertices_to_apply = list(target_weights.keys())
    for i_count, idx in enumerate(vertices_to_apply):
        if cmds.progressWindow(query=True, isCancelled=True):
            cmds.progressWindow(endProgress=True)
            cmds.warning("Transfer cancelled by user.")
            return

        if i_count % 100 == 0:
            progress = int(i_count / len(vertices_to_apply) * 100)
            cmds.progressWindow(edit=True, progress=progress, status=f"Applying weight {i_count+1}/{len(vertices_to_apply)}")
        
        weights = target_weights[idx]
        if np.sum(weights) > 0:
            transform_values = []
            for j, joint in enumerate(target_joints):
                if j < len(weights) and weights[j] > 0.001:
                    transform_values.append((joint, weights[j]))
            
            if transform_values:
                vtx = f"{target_mesh}.vtx[{idx}]"
                try:
                    cmds.skinPercent(target_skin_cluster, vtx, transformValue=transform_values, normalize=True)
                except Exception as e:
                    pass

    cmds.progressWindow(endProgress=True)
    cmds.inViewMessage(amg=f"Weight transfer complete!", pos="midCenter", fade=True)

        self.transfer_btn.setFixedHeight(45)
        self.transfer_btn.setFont(QFont("Arial", 12, QFont.Bold))
        self.transfer_btn.setStyleSheet("QPushButton { background-color: #176517; color: white; border-radius: 5px; } QPushButton:hover { background-color: #2E8B57; }")

    def _create_slider_widget(self, label_text, min_val, max_val, default_val, tooltip):
        widget = QWidget()
        widget.setToolTip(tooltip)
        layout = QGridLayout(widget)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setHorizontalSpacing(8)
        layout.setVerticalSpacing(3)
        label = QLabel(label_text)
        label.setFont(QFont("Arial", 9, QFont.Bold))
        layout.addWidget(label, 0, 0)
        slider = QSlider(Qt.Horizontal)
        slider.setRange(int(min_val * 100), int(max_val * 100))
        slider.setValue(int(default_val * 100))
        slider.setStyleSheet("QSlider::groove:horizontal { border: 1px solid #555; height: 8px; background: #333; margin: 2px 0; border-radius: 4px; } QSlider::handle:horizontal { background: #454545; border: 1px solid #777; width: 18px; margin: -5px 0; border-radius: 5px; }")
        layout.addWidget(slider, 1, 0, 1, 2)
        value_label = QLabel(f"{default_val:.2f}")
        value_label.setMinimumWidth(40)
        value_label.setAlignment(Qt.AlignRight | Qt.AlignVCenter)
        value_label.setFont(QFont("Arial", 9))
        layout.addWidget(value_label, 0, 1)
        def update_label(value):
            actual_value = value / 100.0
            value_label.setText(f"{actual_value:.2f}")
        slider.valueChanged.connect(update_label)
        widget.slider = slider
        widget.value_label = value_label
        return widget

    def _create_int_slider_widget(self, label_text, min_val, max_val, default_val, tooltip):
        widget = QWidget()
        widget.setToolTip(tooltip)
        layout = QGridLayout(widget)
        layout.setContentsMargins(0, 0, 0, 0)

    def get_slider_value(self, slider_widget):
        return slider_widget.slider.value() / 100.0

    def get_int_slider_value(self, slider_widget):
        return slider_widget.slider.value()
        
    def _update_ui_for_mode(self, index):
        mode_data = self.mode_combo.itemData(index)
        if mode_data == 'mesh':
            self.source_btn.setText("Set Source Model")
            self.target_btn.setText("Set Target Model")
            self.vtx_info_label.setVisible(False)
            self.transfer_btn.setText("TRANSFER FULL MESH WEIGHTS")
        elif mode_data == 'vertex':
            self.source_btn.setText("Record Source Vtx")
            self.target_btn.setText("Select Target Vtx")
            self.vtx_info_label.setVisible(True)
            self.transfer_btn.setText("TRANSFER SELECTED VERTEX WEIGHTS")

    def set_source_mesh(self):
        mode_data = self.mode_combo.currentData()
        if mode_data == 'mesh':
            mesh_name, _ = get_selection_info()
            if mesh_name:
                self.source_mesh_le.setText(mesh_name)
                cmds.inViewMessage(amg=f"Source: {mesh_name.split('|')[-1]}", pos="midCenter", fade=True)
            else:
                cmds.warning("Select a source mesh.")
        elif mode_data == 'vertex':
            if record_source_vertices():
                global SOURCE_SELECTION_STATE
                mesh = SOURCE_SELECTION_STATE['mesh']
                count = len(SOURCE_SELECTION_STATE['indices'])
                self.source_mesh_le.setText(mesh.split('|')[-1])
                self.vtx_info_label.setText(f"Source Vtx: {count} Recorded")
                self.vtx_info_label.setStyleSheet("color: #3cb371;")
            else:
                self.vtx_info_label.setText("Source Vtx: None Recorded")
                self.vtx_info_label.setStyleSheet("color: #FFA500;")

    def set_target_mesh(self):
        mode_data = self.mode_combo.currentData()
        # In Mesh mode, set text field. In Vertex mode, it's instructional.
        mesh_name, indices = get_selection_info()
        if not mesh_name:
            cmds.warning("Please select target mesh/vertices.")
            return
            
        if mode_data == 'vertex' and not indices:
             cmds.warning("Switch to component mode and select vertices for target.")
             return

        self.target_mesh_le.setText(mesh_name.split('|')[-1])
        if mode_data == 'vertex':
             cmds.inViewMessage(amg=f"Target Vertices: {len(indices)} selected", pos="midCenter", fade=True)
        else:
             cmds.inViewMessage(amg=f"Target Mesh: {mesh_name.split('|')[-1]}", pos="midCenter", fade=True)

    def run_transfer(self):
        source_mesh = self.source_mesh_le.text()
        target_mesh = self.target_mesh_le.text()
        fidelity = self.get_slider_value(self.fidelity_slider_widget)
        smoothness = self.get_slider_value(self.smoothness_slider_widget)
        max_influences = self.get_int_slider_value(self.max_inf_slider_widget)
        mode = self.mode_combo.currentData()
        
        if mode == 'mesh' and not target_mesh:
             QMessageBox.warning(self, "Error", "请设置目标模型。")
             return

        try:
            global CACHED_DATA
            CACHED_DATA = {}
            
            self.transfer_btn.setEnabled(False)
            self.transfer_btn.setText("Processing...")
            
            transfer_weights_main(source_mesh, target_mesh, fidelity, smoothness, max_influences, mode)
            
        except Exception as e:
            cmds.error(f"Transfer failed: {e}")
            QMessageBox.critical(self, "Error", f"Weight transfer failed: {e}")
        finally:
            self.transfer_btn.setEnabled(True)
            self.transfer_btn.setText("TRANSFER WEIGHTS")
